# Intel Mac（macOS x86_64）专用依赖。
#
# 为什么不能直接用 requirements.txt：
#   unstructured[pdf] 会带上 numba / unstructured-inference / torch，
#   而 2025 年后这些包基本只发布 arm64 的 macOS wheel：
#     - llvmlite（numba 依赖）：macOS wheel 仅 arm64
#     - torch / torchvision：Intel Mac 止步 2.2.2，且与新版 unstructured-inference 版本约束冲突
#     - cryptography 50+ / pikepdf 10.x / opencv-python 5.x：x86_64 wheel 要求 macOS 15+
#   本机为 macOS 14 + Intel，所以固定到仍有 x86_64 wheel 的版本组合。
#
# unstructured 固定 0.18.9：0.19+/0.20+ 依赖 numba（Intel Mac 无法安装）。
# pikepdf 固定 9.x：10.x 的 x86_64 wheel 要求 macOS 15。
# cryptography 由 constraints-macos-intel.txt 约束到 <43（42.0.8 最后支持 macOS 14 x86_64）。

pymupdf>=1.24
unstructured==0.18.9
pdfminer.six                 # unstructured 的 PDF 解析内核
pikepdf>=9.11,<10
pdf2image                    # 页面渲染（hi_res/OCR 用；fast 策略不调用）
pi-heif==1.4.0               # unstructured 导入依赖（最后支持 x86_64 的版本）
opencv-python==4.13.0.92     # unstructured 导入依赖（macosx_14_0_x86_64 wheel）
matplotlib>=3.8,<3.11        # unstructured 分析模块导入依赖
llama-parse>=0.6
